# ClipForge — pinned, CPU-only
# Verified to exist on PyPI. All heavy deps are lazily imported by the
# backend, so `import app` works with just gradio + numpy installed.

# Gradio 4.x is BROKEN on Python 3.14 (codespaces default): its
# safe_get_stop_event() calls asyncio.get_event_loop(), which raises
# RuntimeError on 3.14 (implicit loop creation removed) -> every SSE/queue
# request dies with "Connection errored out." Gradio 6.x fixes this.
gradio>=6.30.0
faster-whisper==1.1.1
# faster-whisper 1.1.1 calls av.open(..., metadata_errors="ignore"), an option
# REMOVED in PyAV >= 12 (it crashes with TypeError). Pin av<12 until
# faster-whisper bumps; PyAV 11.x has wheels for cp310+cp312.
av<12
ctranslate2==4.6.0
onnxruntime==1.20.1
# 4.x REQUIRED: OpenCV 5.0 removed the Haar cascade XMLs and the backend's
# reframe.py relies on Haar cascades (with YuNet/center-crop fallbacks).
opencv-python-headless==4.10.0.84
yt-dlp==2025.10.14
# gradio 4.38+ requires numpy>=2.0, so the whole stack is pinned on numpy 2.x.
numpy==2.0.2
scipy==1.14.1
# HTTP stack verified WITH gradio 6.30.0 (E2E-tested 2026-10-15 — this exact
# set rendered 3 real clips with thumbnails + virality scores): fastapi 0.143.1
# + starlette 1.8.0 were pulled by the gradio 6 upgrade. Do NOT downgrade:
# gradio 6 requires starlette>=1.0.1, and fastapi 0.115.x requires
# starlette<0.42 — the old pins conflict and break fresh installs.
fastapi==0.143.1
starlette==1.8.0
pydantic==2.10.6
# AI dub (beta): translation + neural TTS voices. Both are lazily imported
